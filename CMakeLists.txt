# OFF: el volcado manual del loader usa un buffer fijo de 32 KB que corrompe el heap con los
# shaders grandes de glitch (Asphalt-6-Vita Bugs #009/#010). La cache la hace vitaGL
# (HAVE_SHADER_CACHE=1 + vglSetShaderCachePath en gl_init).
option(DUMP_COMPILED_SHADERS "When using GLSL or CG shader format, cache compiled shaders on disk" OFF)

  if (SHADER_FORMAT IN_LIST valid_dump_formats)
    add_definitions(-DDUMP_COMPILED_SHADERS)
  else()
    message(WARNING "DUMP_COMPILED_SHADERS has no effect for the chosen shader format ${SHADER_FORMAT}")
  endif()
endif()

# vitaGL vendorizado (fuente de Asphalt-6-Vita, que corre glitch + GLES2 con GLSL compilado en
# caliente en consola real). SOFTFP_ABI=1 es obligatorio: libEternalLegacy.so es armeabi
# soft-float y llama a glClearColor/glUniform*/etc. pasando floats en registros r0-r3.
# LOG_ERRORS=1 manda los errores internos de vitaGL a vgl_log() (source/utils/glutil.c).
set(VITAGL_DIR ${CMAKE_SOURCE_DIR}/lib/vitaGL)
set(VITAGL_LIB ${VITAGL_DIR}/libvitaGL.a)
# Flags GLSL del README de vitaGL (el motor manda GLSL ES 1.00 crudo desde shaders.pak):
#   HAVE_GLSL_UBOS / HAVE_GLSL_TEXTURE_SIZE -> extensiones del traductor GLSL
#   HAVE_FIXED_ATTRIBUTES -> atributos GL_FIXED con shaders GLSL
#   SAFE_UNIFORMS -> indexado de arrays de uniforms conforme (glitch sube arrays con glUniform4fv)
#   HAVE_SHARK_LOG -> el log del compilador llega a glGetShaderInfoLog (lo imprime glutil.c)
set(VITAGL_MAKE_FLAGS "SOFTFP_ABI=1 NO_DEBUG=1 LOG_ERRORS=1 HAVE_SHADER_CACHE=1 NO_SPLASHSCREEN=1 DRAW_SPEEDHACK=2 HAVE_GLSL_UBOS=1 HAVE_GLSL_TEXTURE_SIZE=1 HAVE_FIXED_ATTRIBUTES=1 SAFE_UNIFORMS=1 HAVE_SHARK_LOG=1")
separate_arguments(VITAGL_MAKE_ARGS UNIX_COMMAND "${VITAGL_MAKE_FLAGS}")

# Rebuild completo de vitaGL si cambian los flags (la regla OUTPUT sola no lo hace).
set(VITAGL_FLAGS_STAMP ${CMAKE_BINARY_DIR}/vitagl_flags.stamp)
set(vitagl_old_flags "")
if (EXISTS ${VITAGL_FLAGS_STAMP})
  file(READ ${VITAGL_FLAGS_STAMP} vitagl_old_flags)
endif()
if (NOT vitagl_old_flags STREQUAL VITAGL_MAKE_FLAGS)
  file(GLOB_RECURSE vitagl_stale_objs ${VITAGL_DIR}/source/*.o)
  file(REMOVE ${vitagl_stale_objs} ${VITAGL_LIB})
  file(WRITE ${VITAGL_FLAGS_STAMP} "${VITAGL_MAKE_FLAGS}")
endif()
